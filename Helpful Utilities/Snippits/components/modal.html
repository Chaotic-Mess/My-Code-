Modal Component

A centered overlay dialog with a back-drop and close functionality.

×

Hello!

This is a minimalistic modal window.

HTML

<!-- Trigger Button -->
<button id="openBtn">Open Modal</button>

<!-- Modal Overlay -->
<div class="modal-overlay" id="myModal">
    <div class="modal-content">
        <h2>Title</h2>
        <p>Content goes here.</p>
        <button id="closeBtn">Close</button>
    </div>
</div>

CSS

.modal-overlay {
    position: fixed;
    top: 0; left: 0;
    width: 100%; height: 100%;
    background: rgba(0, 0, 0, 0.7);
    display: none;
    justify-content: center;
    align-items: center;
}
.modal-content {
    background: #161b22;
    padding: 2rem;
    border-radius: 12px;
}

JavaScript

const modal = document.getElementById('myModal');
const openBtn = document.getElementById('openBtn');
const closeBtn = document.getElementById('closeBtn');

openBtn.onclick = () => modal.style.display = 'flex';
closeBtn.onclick = () => modal.style.display = 'none';
window.onclick = (e) => {
    if (e.target == modal) modal.style.display = 'none';
}

Full HTML File

<!DOCTYPE html>
<html>
<head>
    <style>
        .modal-overlay { position: fixed; top: 0; left: 0; width: 100%; height: 100%; background: rgba(0,0,0,0.5); display: none; justify-content: center; align-items: center; }
        .modal-content { background: white; padding: 20px; border-radius: 8px; }
    </style>
</head>
<body>
    <button id="openBtn">Open Modal</button>
    <div class="modal-overlay" id="myModal">
        <div class="modal-content">
            <p>Hello World!</p>
            <button id="closeBtn">Close</button>
        </div>
    </div>
    <script>
        const m = document.getElementById('myModal');
        document.getElementById('openBtn').onclick = () => m.style.display = 'flex';
        document.getElementById('closeBtn').onclick = () => m.style.display = 'none';
    </script>
</body>
</html>
