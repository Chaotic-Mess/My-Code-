Form Component

A clean, vertical form layout with styled inputs.

HTML

<form id="singleInputForm">
    <input type="email" id="singleEmailInput" placeholder="you@example.com">
    <button type="submit">Show Data</button>
</form>
<div id="singleFormOutput"></div>

CSS

#singleInputForm { display: flex; flex-direction: column; gap: 1rem; max-width: 300px; }
#singleInputForm input { padding: 10px; border-radius: 6px; border: 1px solid #ccc; }
#singleInputForm button { padding: 10px; background: #58a6ff; color: white; border: none; border-radius: 6px; cursor: pointer; }
#singleFormOutput { margin-top: 1rem; color: #8b949e; }

JavaScript

document.getElementById('singleInputForm').addEventListener('submit', function(e) {
    e.preventDefault();
    const email = document.getElementById('singleEmailInput').value;
    document.getElementById('singleFormOutput').innerHTML = `Email: ${email || '-'}`;
});

HTML

<form id="multiInputForm">
    <input type="text" id="nameInput" placeholder="Your Name">
    <input type="email" id="emailInput" placeholder="you@example.com">
    <input type="text" id="colorInput" placeholder="e.g. Blue">
    <button type="submit">Show Data</button>
</form>
<div id="formOutput"></div>

CSS

#multiInputForm { display: flex; flex-direction: column; gap: 1rem; max-width: 300px; }
#multiInputForm input { padding: 10px; border-radius: 6px; border: 1px solid #ccc; }
#multiInputForm button { padding: 10px; background: #58a6ff; color: white; border: none; border-radius: 6px; cursor: pointer; }
#formOutput { margin-top: 1rem; color: #8b949e; }

JavaScript

document.getElementById('multiInputForm').addEventListener('submit', function(e) {
    e.preventDefault();
    const name = document.getElementById('nameInput').value;
    const email = document.getElementById('emailInput').value;
    const color = document.getElementById('colorInput').value;
    document.getElementById('formOutput').innerHTML = `Name: ${name || '-'}
Email: ${email || '-'}
Favorite Color: ${color || '-'}`;
});
