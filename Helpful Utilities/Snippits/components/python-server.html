A full-stack snippet featuring a Python (Flask) backend and a fetch-based HTML interface.

Server: localhost:5000 [MOCK SERVER]

Python (app.py)

from flask import Flask, request, jsonify
from flask_cors import CORS

app = Flask(__name__)
CORS(app) # Enable CORS for frontend interaction

@app.route('/api/message', methods=['POST'])
def handle_message():
    data = request.json
    user_msg = data.get('message', '')
    return jsonify({
        "status": "success",
        "reply": f"Server received: {user_msg}"
    })

if __name__ == '__main__':
    app.run(debug=True, port=5000)

JavaScript

async function sendMessage() {
    const msg = document.getElementById('userInput').value;
    const responseBox = document.getElementById('response');
    
    try {
        const res = await fetch('http://localhost:5000/api/message', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ message: msg })
        });
        const data = await res.json();
        responseBox.innerText = data.reply;
    } catch (err) {
        responseBox.innerText = "Error: Server not running or unreachable.";
    }
}

Full script.py (Single File App)

from flask import Flask, request, jsonify, render_template_string
from flask_cors import CORS

app = Flask(__name__)
CORS(app)

@app.route('/')
def index():
    return render_template_string("""
        <!DOCTYPE html>
        <html>
        <head>
            <title>Python Server Interface</title>
            <style>
                body { font-family: sans-serif; padding: 2rem; background: #0d1117; color: white; }
                input { padding: 10px; border-radius: 4px; border: 1px solid #30363d; background: #010409; color: white; }
                button { padding: 10px 20px; border-radius: 4px; background: #58a6ff; color: white; border: none; cursor: pointer; }
                #response { margin-top: 20px; color: #8b949e; }
            </style>
        </head>
        <body>
            <h1>Python API Interface</h1>
            <input type="text" id="userInput" placeholder="Message...">
            <button onclick="sendMessage()">Send</button>
            <div id="response">Waiting for action...</div>

            <script>
                async function sendMessage() {
                    const msg = document.getElementById('userInput').value;
                    const resBox = document.getElementById('response');
                    resBox.innerText = 'Connecting...';
                    try {
                        const res = await fetch('/api/message', {
                            method: 'POST',
                            headers: { 'Content-Type': 'application/json' },
                            body: JSON.stringify({ message: msg })
                        });
                        const data = await res.json();
                        resBox.innerText = 'Server says: ' + data.reply;
                    } catch (err) {
                        resBox.innerText = 'Error connecting to server';
                    }
                }
            </script>
        </body>
        </html>
    """)

@app.route('/api/message', methods=['POST'])
def handle_message():
    data = request.json
    return jsonify({"reply": f"Received: {data['message']}"})

if __name__ == '__main__':
    app.run(debug=True, port=5000)
