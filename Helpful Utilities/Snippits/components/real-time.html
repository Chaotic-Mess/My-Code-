Full script.py

from flask import Flask, Response, render_template_string
import time
import random

app = Flask(__name__)

@app.route('/')
def index():
    return render_template_string("""
        <!DOCTYPE html>
        <html>
        <head>
            <title>Real-Time Feed</title>
            <style>
                #display { font-size: 2rem; color: #238636; font-family: monospace; }
            </style>
        </head>
        <body>
            <h1>Live Data</h1>
            <div id="display">Waiting...</div>
            <script>
                const source = new EventSource('/stream');
                source.onmessage = (e) => {
                    document.getElementById('display').innerText = e.data;
                };
            </script>
        </body>
        </html>
    """)

@app.route('/stream')
def stream():
    def generate_data():
        while True:
            yield f"data: {random.randint(10, 99)}\\n\\n"
            time.sleep(1)
    return Response(generate_data(), mimetype='text/event-stream')

if __name__ == '__main__':
    app.run(debug=True, port=5000)

Full script.py (JSON version)

from flask import Flask, Response, render_template_string
import time
import random
import json

app = Flask(__name__)

@app.route('/')
def index():
    return render_template_string("""
        <!DOCTYPE html>
        <html>
        <head>
            <title>Real-Time JSON Feed</title>
            <style>
                #json-output { white-space: pre; color: #58a6ff; padding: 1rem; background: #010409; }
            </style>
        </head>
        <body>
            <h1>Live JSON Stream</h1>
            <div id="json-output">Waiting...</div>
            <script>
                const source = new EventSource('/stream');
                source.onmessage = (e) => {
                    const data = JSON.parse(e.data);
                    document.getElementById('json-output').innerText = JSON.stringify(data, null, 2);
                };
            </script>
        </body>
        </html>
    """)

@app.route('/stream')
def stream():
    def generate_json():
        while True:
            data = {"value": random.randint(100, 999), "status": "ok"}
            yield f"data: {json.dumps(data)}\\n\\n"
            time.sleep(1)
    return Response(generate_json(), mimetype='text/event-stream')

if __name__ == '__main__':
    app.run(debug=True, port=5000
